配方测试工步树加载修复逻辑流程图

        flowchart TD
            Start["用户选择配方 (recipe_tree 单击/双击)"] --> Trigger["触发 load_recipe_to_tree()"]
            Trigger --> Clear["清空当前工步树 self.step_tree.clear()"]
            Clear --> ParseInfo["读取配方 JSON 参数 (CAN波特率, 匹配正则等)"]
            ParseInfo --> LoopItems{"遍历配方中的 items 主工步"}
            
            LoopItems -- "有未处理主工步" --> BuildParent["创建主工步 QTreeWidgetItem"]
            BuildParent --> SetParentMeta["设置主工步 UserRole 元数据 (standard_type, retry_count, unit)"]
            SetParentMeta --> LoopSub{"遍历 sub_steps 子工步"}
            
            LoopSub -- "有未处理子工步" --> BuildChild["创建子工步 QTreeWidgetItem"]
            BuildChild --> SetChildMeta["设置子工步 UserRole 元数据 (device, type, is_judgment, fail_strategy)"]
            SetChildMeta --> AddChild["parent.addChild(child)"]
            AddChild --> LoopSub
            
            LoopSub -- "子工步遍历完毕" --> AddParent["self.step_tree.addTopLevelItem(parent)"]
            AddParent --> LoopItems
            
            LoopItems -- "所有主工步遍历完毕" --> Expand["self.step_tree.expandAll() 展开所有工步"]
            Expand --> End["配方测试工步正常展示完毕"]
