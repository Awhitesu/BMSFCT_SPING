    InitRetry["初始化菊花链自愈次数 daisy_retry = 0"]

    Delay10["步骤2: 延时等待 10 秒"]

    Start --> InitRetry
    InitRetry --> StepA
    StepA --> StepB
    StepB -- "是" --> ExecB
    ExecB --> StepC
    StepB -- "否" --> StepC
    StepC -- "是" --> ExecC
    ExecC --> CheckFive
    StepC -- "否" --> CheckFive
    
    CheckFive -- "否 (正常)" --> FinalEval
    CheckFive -- "是 (接触不良)" --> CheckDaisyLimit
    
    CheckDaisyLimit -- "是" --> IncRetry
    IncRetry --> ActionDisconnect
    ActionDisconnect --> Delay10
    Delay10 --> ActionConnect
    ActionConnect --> ActionConfigNode
    ActionConfigNode --> Delay5
    Delay5 --> StepA
    
    CheckDaisyLimit -- "否 (超过3次)" --> FailDaisy
