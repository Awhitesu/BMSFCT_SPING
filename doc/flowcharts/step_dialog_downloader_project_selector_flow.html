flowchart TD
    Start["打开编辑指令弹窗 (StepDialog)"] --> CheckDevice{"控制设备是否为力捷丰 SG400 ?"}
    
    CheckDevice -- "否" --> OtherDevice["渲染其他常规设备页面"]
    CheckDevice -- "是" --> CheckAction{"功能动作是否为执行烧录工程 ?"}
    
    CheckAction -- "否 (连接检测/查询状态)" --> ShowEmpty["显示 Page 14 (无需额外参数)"]
    CheckAction -- "是" --> ShowOpteeqPage["显示 Page 11 (力捷丰烧录配置)"]
    
    ShowOpteeqPage --> InitProjectSelector["初始化工程选择器 (可编辑下拉框 combo_project)"]
    InitProjectSelector --> CheckDeviceMgr{"DeviceManager/下载器连接是否就绪 ?"}
    
    CheckDeviceMgr -- "已连接/就绪" --> QueryRemote["自动拉取下载器工程列表 (/home/s2g/proj/)"]
    CheckDeviceMgr -- "未连接/离线" --> ScanLocalCache["扫描本地工程缓存目录 (cache/projects)"]
    
    QueryRemote --> PopulateCombo["工程列表填充至下拉选择框"]
    ScanLocalCache --> PopulateCombo
    
    PopulateCombo --> UserInteraction{"用户交互方式"}
    
    UserInteraction -- "点击下拉框直接选择" --> SelectFromList["选中下载器已有脱机工程 (如 UnlockHSMCode.opj)"]
    UserInteraction -- "点击 [🔄 读取下载器工程]" --> ManualFetch["触发在线实时读取指令 (channel:getfilelist)"]
    UserInteraction -- "点击 [📂 本地文件]" --> SelectLocalFile["调起 Windows 文件对话框选择本地 .opj/.oprj"]
    UserInteraction -- "直接输入" --> CustomInput["输入自定义工程名称"]
    
    ManualFetch --> CheckFetchResult{"读取下载器工程是否成功 ?"}
    CheckFetchResult -- "成功" --> UpdateComboSuccess["刷新并高亮展示全部脱机工程"]
    CheckFetchResult -- "失败/超时" --> FallbackCache["告警提示并回退保留本地缓存清单"]
    
    SelectFromList --> ConfirmParam["设定完成超时时间 (SpinBox)"]
    UpdateComboSuccess --> ConfirmParam
    FallbackCache --> ConfirmParam
    SelectLocalFile --> ConfirmParam
    CustomInput --> ConfirmParam
    
    ConfirmParam --> ClickOK["点击 [确认指令]"]
    ClickOK --> GetData["get_data() 拦截并提取 PROJECT 与 TIMEOUT"]
    GetData --> OutputRecipe["输出测试工步数据 (params: PROJECT:xxx / TIMEOUT:yyy)"]
    OutputRecipe --> End["成功保存工步并更新测试树"]
